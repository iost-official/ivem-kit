# useConfig

## Usage

```tsx
import { useConfig } from '@ivem/kit-react'

const config = useConfig()
```

## Parameters

- None

## Return Type

- `Config`

## Examples

```tsx
function DebugConfig() {
  const config = useConfig()
  return <pre>{JSON.stringify(config.state, null, 2)}</pre>
}
```
