# useAccount

## Usage

```ts
import { useAccount } from '@ivem/kit-vue'

const account = useAccount()
```

## Parameters

- None

## Return Type

- `Ref<GetAccountReturnType>`

## Examples

```ts
if (account.value.isConnected) {
  console.log(account.value.address)
}
```
